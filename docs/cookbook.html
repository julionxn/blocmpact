    19. Cookbook

    Each family of effects, with the pieces that tend to work. The numbers are starting points to tune.

    Family Blocmpact pieces
    Hits and impacts Shape.arc/sphere + Velocity.along, a short sharp sound, ScreenFx.shake/flash for 2–4 ticks, impact only on finishing blows
    Projectiles and trails Shape.line between positions, Mesh.ribbon for the trail, beamGlow for the core
    Rays and lasers beamGlow + lightReach to tint the ground, pointGlow at the contact point
    Lightning lightningGlow, a bluish flash, sounds().atSpeedOfSound for the thunder, env.thunder
    Explosions and shockwaves Velocity.towards (suction beforehand), frontRing + frontRefract, a dust marchVolume, impactCycle
    Auras and buffs Slow Shape.helix/ring, Mesh.ring, entities().outline, a pulse with FxMath.wave
    Altered states An overlay or scene pass with vignetteMask, wobble, heatHaze; compositor saturation, tint, vignette
    Shields and barriers sphereShell + fresnel + hexGrid; a hit ripple with onEvent("hit")
    Portals and teleporting swirl, pixelate, a particle spiral, a flash when crossing
    Magic circles decalUV/planeUV + glyphBand, sdPolygram, ticks, revealAround
    Weather and ambience EnvironmentFx (sky, fog, light, rain, clouds) + ambient particles with Shape.box
    Transforming the world The same radius function in ServerEffect (blocks) and in the shader (frontRing, cracks with isoLine and blackbody)
    Diegetic interface Canvas (viewfinder, reticle, marker), a 7-segment readout on the terrain, scanlines, depthEdges × distanceFront
    Cinematic moments Cutscene over time(), silencing, holdFps(12) + frozenWorld while the boom travels

    Full example in Java

    This effect uses every layer at once and was tested in game (the screenshots showed the beam, the refracting shockwave, the fog and night sky, the flash, the impact frame sequence, the HUD, and the player's outline and pose).

    public final class ShowcaseFx extends VisualEffect {
    static final int IMPACT = 20, IMPACT_FRAMES = 60, END = 200;
    private static final Identifier PILLAR = Identifier.fromNamespaceAndPath("mymod", "pillar");
    private final int color;
    private final FxParticleOptions spark;

    public ShowcaseFx(EffectSpawn spawn) {
        super(spawn);
        this.color = params().getColor("color", 0xFF66CCFF);
        this.spark = FxParticles.options().color(color).size(0.15f, 0.02f).lifetime(30).gravity(-0.002f).additive()
            .sprite(FxParticles.SOFT).build();
    }

    @Override public int duration() { return END; }

    private float k(float t) { return FxMath.window(t, 0, 10, END - 30, END); }

    @Override protected void onTick(TickContext ctx, int t) {
        Vec3 o = origin();
        if (t % 2 == 0) ctx.particles().emit(spark, Shape.ring(o, 3, t * 0.1), 24, Velocity.fixed(0, 0.05, 0));
        if (t == IMPACT) {
            ctx.particles().emit(ParticleTypes.END_ROD, Shape.sphere(o.add(0, 1, 0), 0.5), 120, Velocity.along(0.2, 0.5));
            ctx.sounds().at(SoundEvents.GENERIC_EXPLODE, o, 2f, 0.6f);
        }
    }

    @Override public void passes(Frame f, PassCollector out) {
        float t = time(f.partial()), since = Math.max(0, since(IMPACT, f.partial()));
        Vec3 rel = f.rel(origin(f.partial()));
        out.add(PILLAR, u -> { u.v3("Target", rel); u.f4("P0", 3f, since, seedF(), k(t)); u.color("P1", color); });
    }

    @Override public void screen(Frame f, ScreenFx s) {
        float since = since(IMPACT, f.partial());
        s.bloom(0.8f, 1.2f);
        s.vignette(0.4f * k(time(f.partial())), 0f, 0f, 0f);
        if (since >= 0) {
            s.flash(1.4f * (float) Math.exp(-7 * since), 1, 1, 1);
            s.shake(0.6f * (float) Math.exp(-3 * since));
        }
        Vector3f focus = f.projectUv(origin(f.partial()));
        s.impactCycle(since(IMPACT_FRAMES, f.partial()), 0.1f, focus.x, focus.y,
            ImpactFrame.builder(ImpactFrame.Mode.TWO_TONE).build(),
            ImpactFrame.builder(ImpactFrame.Mode.HALFTONE).paper(0xFFF4EBD9).ink(0xFF201010).build(),
            ImpactFrame.builder(ImpactFrame.Mode.RAINBOW).build());
    }

    @Override public void environment(Frame f, EnvironmentFx env) {
        float k = k(time(f.partial()));
        env.fogColor(0x301040, k).fogDistance(0, 48, k).skyColor(0x100818, k).starBrightness(1f, k);
        env.cloudHole(origin().x, origin().z, 40, 0.3f, seed());
    }

    @Override public void camera(Frame f, CameraFx cam) {
        float since = since(IMPACT, f.partial());
        if (since >= 0) cam.shake(2f * (float) Math.exp(-4 * since));
    }

    @Override public void entities(Frame f, EntityFx e) {
        var player = Minecraft.getInstance().player;
        if (player == null) return;
        e.outline(player, color);
        e.pose(player, EntityFx.RIGHT_ARM, PartPose.rotation(-2.6f, 0, 0).withWeight(k(time(f.partial()))));
    }

    @Override public void geometry(GeometryContext g) {
        float p = g.frame().partial(), t = time(p);
        Vec3 o = origin(p);
        int glow = FxMath.fadeAlpha(color, k(t));
        g.draw(FxRenderTypes.additive(FxTextures.SOFT), m -> m.billboard(o.add(0, 6, 0), 2.5f, t * 0.05f, glow));
        g.draw(FxRenderTypes.additive(), m -> m.ring(o.add(0, 0.05, 0), new Vec3(0, 1, 0), 4.5f, 5f, 64, glow, glow & 0x00FFFFFF));
    }

    @Override public void hud(HudContext h) {
        float t = time(h.partial());
        try (Canvas c = h.canvas(720)) {
            float a = k(t);
            c.spaced("SHOWCASE", c.width() / 2, 60, 3f, FxMath.lerp(4, 1, FxMath.smooth(t / 10f)), FxMath.withAlpha(0xFFFFFF, a));
            Vector3f p = c.project(origin(h.partial()));
            if (p.z > 0) c.reticle(p.x, p.y, FxMath.lerp(60, 14, FxMath.smooth(t / 8f)), 8, 2, FxMath.withAlpha(color, a));
        }
    }
}

    Advanced example: feedback trail and an echo of the past

    out.add(PILLAR, u -> { u.world("Target", movingPosition); u.f4("P0", 2f, 0f, seedF(), 1f); u.f4("P1", 1f, 0.5f, 0.2f, 1f); });
out.add(PostPass.of(TRAIL_WRITE, u -> { u.target("Prev", "mymod_trail"); u.f1("P0", 0.92f); }).toPersistent("mymod_trail"));
out.add(PostPass.of(TRAIL_APPLY, u -> { u.target("Trail", "mymod_trail"); u.history("Past", 1.0f); }).define("GHOST"));

    // trail_write.fsh: keeps what is bright and lets it fade
uniform sampler2D Prev;
void main() {
    vec3 prev = texture(Prev, texCoord).rgb * P0.x;
    vec3 now = texture(DiffuseSampler, texCoord).rgb;
    fragColor = vec4(max(prev, now * smoothstep(0.7, 0.95, lum(now))), 1.0);
}

    // trail_apply.fsh: adds the trail and, with GHOST, a ghost of the image from one second ago
uniform sampler2D Trail;
uniform sampler2D Past;
void main() {
    vec3 c = texture(DiffuseSampler, texCoord).rgb;
#ifdef GHOST
    c = mix(c, texture(Past, texCoord).rgb, 0.35);
#endif
    fragColor = vec4(c + texture(Trail, texCoord).rgb * 0.8, 1.0);
}
